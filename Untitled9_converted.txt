import seaborn as sns
import matplotlib.pyplot as plt

for state in df["state"].unique():

    sns.lmplot(
        data=state_data,
        x="cases",
        y="deaths",
        height=5,
        aspect=1.3
    )

    plt.title(f"Cases vs Deaths - {state}")
